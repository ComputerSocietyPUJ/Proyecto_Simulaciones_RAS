Gracias al patrón Strategy & Registry implementado en static/js/simulacion.js, puedes añadir cualquier nuevo modelo físico o simulación sin modificar el bucle de renderizado: 

// 1. Creas tu clase cumpliendo con el contrato (init, update, draw, reset)
class MiNuevaSimulacion {
    constructor() {
        this.nombre = "Control de Trayectoria PID";
        this.descripcion = "Seguimiento de línea o waypoints con controlador PID.";
        this.reset();
    }
    reset() { /* Inicializar variables de estado */ }
    update(dt, input, engine) { /* Física, ecuaciones diferenciales o sensores */ }
    draw(ctx, engine) { /* Dibujar elementos en el canvas 2D */ }
    getTelemetry() { return { x: 10, y: 20, velocidad: "40 px/s" }; }
}

// 2. La registras en el catálogo (aparecerá automáticamente en el selector)
SimulationRegistry.register('pid_seguidor', {
    name: 'Control de Trayectoria PID',
    create: () => new MiNuevaSimulacion()
});
